Elastic net cox regression

This is less sensitive to noise & overfitting. alpha = 1 for lasso, alpha = 0 for ridge

Ridge (L2), alpha = 0: Useful when you suspect all variables may be relevant but need regularization. Lasso (L1), alpha = 1: Helps with feature selection by pushing less informative coefficients to zero, simplifying the model.

# Define a sequence of alpha values to test
alpha_values <- seq(0, 1, by = 0.1)  # Adjust step size as needed

# Initialize a data frame to store results
elasticnet_cvresults <- data.frame(alpha = numeric(), lambda_min = numeric(), cvm_min = numeric(), outcome = character())

for (outcome in outcomes){
  
  ## Add CD6 and others for CD4 outcome, since these were not significant for CD4 outcome
  if ("CD6" %notin% colnames(train[[outcome]])){
    dataset_train <- train[[outcome]] %>% filter(!is.na(status)) %>% left_join(
      train[["VL4K"]] %>% select(ID, CD6, SH2D1A, CNGB3), by = "ID")
    dataset_test <- test[[outcome]] %>% filter(!is.na(status)) %>% left_join(
      test[["VL4K"]] %>% select(ID, CD6, SH2D1A, CNGB3), by = "ID")
  }else{
    dataset_train <- train[[outcome]]
    dataset_test <- test[[outcome]]
  }
  
  # Loop through each alpha value
  for (alpha in alpha_values) {
    # Perform cross-validation for each alpha
    set.seed(123)  # For reproducibility
    
    penalizedmodel_balanced <- GLMnet(formula_model_small, data = dataset_train, alpha = alpha, lambda = NULL, cv = TRUE, nfolds = 10, standardize = TRUE)
  
    # Store the results: best lambda and corresponding CV error
    elasticnet_cvresults <- rbind(
      elasticnet_cvresults, 
      data.frame(alpha = alpha,
                 lambda_min = penalizedmodel_balanced$fit$lambda.min,
                 cvm_min = min(penalizedmodel_balanced$fit$cvm),
                 outcome = outcome))
  }
}

obtain_scores <- function(models_list, dataset, survformula, traintest){
  score_result <- Score(models_list,
        formula = survformula,
        data = dataset,
        metrics = "auc",
        summary = "risks",
        times = times,
        plots = "ROC")
  
  AUC_overtime <- score_result$AUC$score
  AUC_overtime$type <- traintest

  ROC_overtime <- score_result$ROC$plotframe
  ROC_overtime$type <- traintest
  
  AUCROC_overtime <- list(AUC_overtime = AUC_overtime, ROC_overtime = ROC_overtime)

  return(AUCROC_overtime)
}

# Define function to get the AUC over time
obtain_aucroc <- function(outcome, metric){
  if (metric %notin% c("AUC", "ROC")) {
    stop("Invalid metric! Metric must be either 'AUC' or 'ROC'")
  }
  if (outcome %notin% c("VL4K", "HYBRID", "CD4")) {
    stop("Invalid outcome! Metric must be one of: VL4K, HYBRID, CD4")
  }
  
  # Add CD6 and others for CD4 outcome, since these were not significant for CD4 outcome
  if ("CD6" %notin% colnames(train[[outcome]])){
    dataset_train <- train[[outcome]] %>% filter(!is.na(status)) %>% left_join(
      train[["VL4K"]] %>% select(ID, CD6, SH2D1A, CNGB3), by = "ID")
    dataset_test <- test[[outcome]] %>% filter(!is.na(status)) %>% left_join(
      test[["VL4K"]] %>% select(ID, CD6, SH2D1A, CNGB3), by = "ID")
  }else{
    dataset_train <- train[[outcome]]
    dataset_test <- test[[outcome]]
  }

  # Train the cox model
  formula_model_small <- as.formula(paste("Surv(time, status) ~", paste(c(clinicalpredictors, proteomic_selection), collapse = " + ")))

  coxmodel <- coxph(formula_model_small, dataset_train, x = TRUE)
  
  # Define the formula for other models
  features <- train[[outcome]] %>% select(-c(time, status, ID)) %>% colnames()
  formula_model <- as.formula(paste("Surv(time, status) ~", paste(features, collapse = " + ")))

  # Train the elastic-net (balanced) penalized cox model
  elasticnetcox_model <- GLMnet(
    formula = formula_model,
    data = dataset_train,
    alpha = 1,
    lambda = NULL,
    cv = TRUE,
    nfolds = 10,
    standardize = TRUE)

  # train the RSF, random survival forest
  rsfmodel <- rfsrc(formula_model, data = dataset_train, ntree = 3000, cv.fold = 10, nodesize = 15, mtry = 4, splitrule = "logrankscore", nsplit = 10)

  # Fit a GBM model with survival data
  gbm_model <- obtain_gbmmodel(outcome = outcome, 
                               dataset_train = dataset_train)

  # Calculate AUC data over time for train and test
  survformula <- as.formula(paste("Surv(time, status) ~ 1"))
  model_list <- list("Cox" = coxmodel, 
                     "RSF" = rsfmodel,
                     "GBM" = gbm_model,
                     "ElasticNetCox" = elasticnetcox_model)
  auc_train <- obtain_scores(model_list, dataset_train, survformula, "train")$AUC_overtime
  auc_test <-  obtain_scores(model_list, dataset_test, survformula, "test")$AUC_overtime

  auc_combined <- rbind(auc_train, auc_test)
  auc_combined$outcome <- outcome
  
  roc_train <- obtain_scores(model_list, dataset_train, survformula, "train")$ROC_overtime
  roc_test <-  obtain_scores(model_list, dataset_test, survformula, "test")$ROC_overtime

  roc_combined <- rbind(roc_train, roc_test)
  roc_combined$outcome <- outcome

  if (metric == "AUC"){
    return(auc_combined)
  }else if (metric == "ROC"){
    return(roc_combined)
  }else{return(NA)}
}
